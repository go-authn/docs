{{- /* Navbar version selector, as in claimward/docs. Each documentation version is a separate build published under //, where is the documentation version (params.authn.version in hugo.yaml, as MAJOR.MINOR: 0.1, ...), and its baseURL ends with that version. A build cannot know the versions published after it, so the server renders only the current one; in the browser the list is filled from /versions.json, which the deploy job rewrites in the format mike writes (version, title, aliases), the format go-fileshare's selector reads too. The site is ready for several languages: English at the root of a version, the others under their code (/fr/...). Each option opens, in that version, the first of these that exists: the current page in the current language; the same page in English; the home in the current language; the version's home. A page that a version does not have therefore never ends on a 404. The language prefix is read from the home's permalink, not guessed from the URL. Built on Hextra's own navbar dropdown (assets/js/core/nav-menu.js), so its keyboard handling applies. */ -}} {{- $page := . -}} {{- $basePath := strings.TrimSuffix "/" (urls.Parse site.BaseURL).Path -}} {{- $current := path.Base $basePath -}} {{- $docsRoot := printf "%s/" (strings.TrimSuffix "/" (path.Dir $basePath)) -}} {{- $versionRoot := printf "%s%s/" $docsRoot $current -}} {{- $rest := strings.TrimPrefix $versionRoot $page.RelPermalink -}} {{- /* "fr/" in French, "" in English (the default language, at the root). */ -}} {{- $langPrefix := strings.TrimPrefix $versionRoot site.Home.RelPermalink -}} {{- /* The same page in the default language: same file names in every content//. */ -}} {{- $restDefault := cond (ne $langPrefix "") (strings.TrimPrefix $langPrefix $rest) $rest -}} {{- $dev := site.Params.versions.development | default "dev" -}} {{- $itemClass := "hx:text-gray-600 hx:hover:text-gray-800 hx:dark:text-gray-400 hx:dark:hover:text-gray-200 hx:relative hx:cursor-pointer hx:whitespace-nowrap hx:rounded-sm hx:py-1.5 hx:transition-colors hx:ltr:pl-3 hx:ltr:pr-9 hx:rtl:pr-3 hx:rtl:pl-9 hx:flex hx:items-center hx:gap-1 hx:hover:bg-gray-100 hx:dark:hover:bg-neutral-800" -}} {{- $iconAttrs := `height="1em" class="hx:inline-block"` -}}
